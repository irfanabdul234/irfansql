CREATE PROCEDURE INSERTGMAILS (@INSFIRSTNAME VARCHAR (200), @INSLASTNAME VARCHAR (200), @INSNUMBER VARCHAR (200), @INSDEPID INT, @INSGMAIL VARCHAR (200) = NULL,
@INSDOF DATETIME)
AS
BEGIN 
DECLARE @INSFIINALGMAIL VARCHAR (200)
IF @INSGMAIL IS NULL OR LTRIM(RTRIM(@INSGMAIL)) = ''
BEGIN 
SET @INSFIINALGMAIL = LOWER(@INSFIRSTNAME + @INSLASTNAME + '@gmail.com')
END 
ELSE 
BEGIN 
SET @INSFIINALGMAIL = @INSGMAIL
END 
 
 IF NOT EXISTS (
 SELECT 1 FROM EMPLOYEE
 WHERE FIRSTNAME = @INSFIRSTNAME) AND @INSDOF <= DATEADD(YEAR, -6, GETDATE())
 BEGIN 
 INSERT INTO EMPLOYEE (FIRSTNAME, LASTNAME, PHONENUMBER, DEPARTMENTID, GMAIL, DATEOFBIRTH)
 VALUES (@INSFIRSTNAME, @INSLASTNAME, @INSNUMBER, @INSDEPID, @INSGMAIL, @INSDOF)
 END 
 ELSE 
 BEGIN 
 PRINT 'WE CANNOT PROCEED: EITHER UNDERAGE OR CUSTOMER ALREADY EXIST'
 END 
		END
